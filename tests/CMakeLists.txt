cmake_minimum_required(VERSION 3.18)
project(GPU2VectorTests)

# Find Dependencies
find_package(OpenMP REQUIRED)
find_package(MLIR REQUIRED CONFIG)

# Find necessary binaries
find_program(CGEIST_BIN cgeist REQUIRED)
find_program(MLIR_OPT_BIN mlir-opt REQUIRED)
find_program(MLIR_TRANS_BIN mlir-translate REQUIRED)

# Define Paths
set(CUDA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/kernels.cu")
set(KERNEL_MLIR "${CMAKE_CURRENT_BINARY_DIR}/kernels.mlir")
set(KERNEL_LL "${CMAKE_CURRENT_BINARY_DIR}/kernels.ll")
set(KERNEL_OBJ "${CMAKE_CURRENT_BINARY_DIR}/kernels.o")

# Access the plugin built in the parent directory
# $<TARGET_FILE:CudaVectorPass> is a generator expression that gets the path to the .so
set(PASS_PLUGIN "$<TARGET_FILE:CudaVectorPass>")

# The MLIR Transformation Pipeline
# This command runs cgeist -> mlir-opt -> mlir-translate -> clang
add_custom_command(
    OUTPUT ${KERNEL_OBJ}
    COMMAND ${CGEIST_BIN} ${CUDA_SRC} --cuda-lower -S -o ${KERNEL_MLIR}
    COMMAND ${MLIR_OPT_BIN} ${KERNEL_MLIR}
            -load-pass-plugin=${PASS_PLUGIN}
            --cuda-to-vector="reg-bit-width=256"
            --convert-scf-to-openmp
            --convert-vector-to-scf
            --convert-vector-to-llvm
            --finalize-memref-to-llvm
            --convert-func-to-llvm="emit-c-wrappers"
            --reconcile-unrealized-casts
            -o ${CMAKE_CURRENT_BINARY_DIR}/vectorized.mlir
    COMMAND ${MLIR_TRANS_BIN} --mlir-to-llvmir ${CMAKE_CURRENT_BINARY_DIR}/vectorized.mlir -o ${KERNEL_LL}
    COMMAND ${CMAKE_CXX_COMPILER} -x ir ${KERNEL_LL} -c -o ${KERNEL_OBJ} 
            -O3 -march=native ${OpenMP_CXX_FLAGS}
    DEPENDS ${CUDA_SRC} CudaVectorPass
    COMMENT "Building vectorized CPU kernel from CUDA source"
)

# Build the Test Suite Executable
add_executable(test_suite 
    main.cpp 
    ${KERNEL_OBJ} # Link the object generated by the custom command above
)

# Link Libraries
# - OpenMP for multi-core distribution
# - m for the math library (required for sin, exp, etc.)
target_link_libraries(test_suite 
    PRIVATE 
    OpenMP::OpenMP_CXX 
    m
)

add_test(NAME GPU2VectorTestSuite COMMAND test_suite)
